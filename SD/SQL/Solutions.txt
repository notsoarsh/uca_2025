WHERE EXISTS (
    SELECT 1
    FROM Rating r1
    JOIN Rating r2
      ON r1.rID = r2.rID
     AND r1.mID = r2.mID
    WHERE r1.rID = rv.rID
      AND r1.mID = m.mID
      AND r2.ratingDate > r1.ratingDate
      AND r2.stars > r1.stars
);
